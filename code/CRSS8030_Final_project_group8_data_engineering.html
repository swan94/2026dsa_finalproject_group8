Introduction

This script prepares the data for the final project for machine learning.

Goals

  • Inspection of the structure and contents of the training and testing files
  • Cleaning of inconsistencies between files, including standardization of site names and previous crop labels
  • Evaluate and manage missing data
  • Add external data from Daymet
  • Merge trait, metadata, and soil data into usable files for modeling in both training and testing sets
  • Export cleaned and organized files to support partner’s modeling workflow
  • Document each major data engineering and merging step for reproducibility

Helper Functions

These helper functions apply the same cleaning rules across training and testing files so that site names, crop labels, and summary calculations are consistent before merging.

# Standardize site names across files
clean_site <- function(x) {
  x %>%
    str_squish() %>% # removes extra spaces at the beginning and end
    str_replace("_\\d{4}$", "") %>%       # remove _2018 style suffix from soil files
    str_replace("^IAH1[a-c]$", "IAH1") %>% # naming consistency
    str_replace("^TXH1-\\s+", "TXH1-") %>% # naming consistency
    str_replace("^MOH1-\\s*rep\\s*[12]$", "MOH1") %>% # naming consistency
    str_replace("^MOH1_[12]$", "MOH1") # naming consistency
}

# Standardize previous crop labels
clean_previous_crop <- function(x) {
  x <- x %>%
    str_squish() %>%
    str_to_lower() %>%
    na_if("")

  case_when(
    x == "soybeans" ~ "soybean", # naming standardization for previous crop planted
    x == "winter wheat" ~ "wheat", # naming standardization for previous crop planted
    TRUE ~ x
  ) %>%
    str_to_title()
}

# Mean that returns NA if all values are missing
mean_na <- function(x) {
  if (all(is.na(x))) { 
    NA_real_   # If all values are missing, it returns NA, otherwise it will calculate the mean and ignore missing values
  } else {
    mean(x, na.rm = TRUE)
  }
}

# Character mode that returns NA if all values are missing
mode_chr <- function(x) {
  x <- x[!is.na(x)]
  if (length(x) == 0) return(NA_character_) # Removes missing values
  ux <- unique(x)
  ux[which.max(tabulate(match(x, ux)))]
}

# Minimum date that returns NA if all values are missing
min_date_na <- function(x) {
  if (all(is.na(x))) {
    as.Date(NA)
  } else {
    min(x, na.rm = TRUE)
  }
}

# Maximum date that returns NA if all values are missing
max_date_na <- function(x) {
  if (all(is.na(x))) {
    as.Date(NA)
  } else {
    max(x, na.rm = TRUE)
  }
}

# Removal of duplicate join columns

remove_join_duplicates <- function(df) {
  y_cols <- names(df)[str_detect(names(df), "\\.y$")]
  
  for (y_col in y_cols) {
    base_name <- str_remove(y_col, "\\.y$")
    x_col <- paste0(base_name, ".x")
    
    if (x_col %in% names(df)) {
      df[[y_col]] <- NULL
    }
  }
  
  names(df) <- str_replace(names(df), "\\.x$", "")
  return(df)
}

remove_all_na_columns <- function(df) {
  df %>% select(where(~ !all(is.na(.))))
}

move_response_first <- function(df, response_col = "yield_mg_ha") {
  if (response_col %in% names(df)) {
    df <- df %>% relocate(all_of(response_col))
  }
  return(df)
}

clean_model_dataset <- function(df, response_col = "yield_mg_ha") {
  df %>%
    remove_join_duplicates() %>%
    remove_all_na_columns() %>%
    clean_names() %>%
    move_response_first(response_col = response_col)
}

align_train_test <- function(train_df, test_df, response_col = "yield_mg_ha") {
  train_predictors <- setdiff(names(train_df), response_col)
  test_predictors <- names(test_df)
  
  common_predictors <- intersect(train_predictors, test_predictors)
  
  train_out <- train_df %>%
    select(any_of(c(response_col, common_predictors)))
  
  test_out <- test_df %>%
    select(any_of(common_predictors))
  
  return(list(train = train_out, test = test_out))
}
